74 private void AplicarFiltro()
75 {
76 // Si no hay lista cargada todavia, no hacemos nada.
77 if (_todasInstalaciones == null) return;
78
79 // Obtenemos el texto de busqueda en minusculas para comparacion sin distincion de mayusculas.
80 var texto = (txtBuscar.Text ?? string.Empty).Trim().ToLower();
81
82 // Si no hay texto mostramos todas las instalaciones sin filtrar.
83 if (string.IsNullOrEmpty(texto))
84 {
85 dgInstalaciones.ItemsSource = _todasInstalaciones; // Mostramos todas.
86 txtTotalInstalaciones.Text = $"{_todasInstalaciones.Count} instalaciones"; // Contador total.
87 return;
88 }
89
90 // Filtramos por nombre de instalacion o nombre del tipo que contengan el texto buscado.
91 var filtradas = _todasInstalaciones.FindAll(i =>
92 (i.Nombre ?? string.Empty).ToLower().Contains(texto) ||
93 (i.TiposInstalacion?.Nombre ?? string.Empty).ToLower().Contains(texto));
94
95 dgInstalaciones.ItemsSource = filtradas; // Mostramos el resultado filtrado.
96 txtTotalInstalaciones.Text = $"{filtradas.Count} instalaciones"; // Contador filtrado.

121 private Instalaciones CrearEntidadDesdeFormulario()
122 {
123 var i = _instalacionSeleccionada ?? new Instalaciones(); // Reutilizamos en edicion.
124
125 i.Nombre = (txtNombre.Text ?? string.Empty).Trim(); // Nombre.
126
127 i.TipoInstalacionId = cmbTipoInstalacion.SelectedValue is int ? (int)cmbTipoInstalacion.SelectedValue : 0; // Tipo.
128
129 i.Disponible = chkDisponible.IsChecked == true; // Disponible.
130
131 i.Observaciones = string.IsNullOrWhiteSpace(txtObservaciones.Text) ? null : txtObservaciones.Text.Trim(); // Observaciones.
132
133 // Parseamos precio (coma o punto).
134 var precioTexto = (txtPrecioHora.Text ?? string.Empty).Trim(); // Texto.
135
136 if (decimal.TryParse(precioTexto, NumberStyles.Any, CultureInfo.CurrentCulture, out var precio) ||
137 decimal.TryParse(precioTexto, NumberStyles.Any, CultureInfo.InvariantCulture, out precio))
138 {
139 i.PrecioHora = precio; // Asignamos.
140 }
141 else
142 {
143 i.PrecioHora = -1; // Forzamos error de validacion.
144 }
145
146 return i; // Devolvemos.
147 }

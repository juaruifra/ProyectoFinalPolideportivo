207 private double CalcularPorcentajeOcupacion(List<ProyectoFinal.model_new.Reservas> reservasHoy)
208 {
209 // Obtenemos el total de instalaciones disponibles del club.
210 var instalaciones = _instalacionesController.ObtenerTodos(soloDisponibles: false);
211 int numInstalaciones = instalaciones.Count; // Total de instalaciones.
212
213 // Si no hay instalaciones no podemos calcular, devolvemos 0.
214 if (numInstalaciones == 0) return 0;
215
216 // Calculamos las horas que el club esta abierto al dia.
217 double horasApertura = (ConfiguracionClub.HoraCierre - ConfiguracionClub.HoraApertura).TotalHours;
218
219 // Capacidad maxima del club en horas (todas las instalaciones todo el dia).
220 double capacidadTotal = numInstalaciones * horasApertura;
221
222 // Sumamos la duracion en horas de cada reserva de hoy.
223 double horasReservadas = reservasHoy.Sum(r =>
224 (r.FechaHoraFin - r.FechaHoraInicio).TotalHours);
225
226 // Calculamos el porcentaje redondeado.
227 return (horasReservadas / capacidadTotal) * 100.0;
228 }

160 private Cuotas CrearEntidadDesdeFormulario()
161 {
162 var c = _cuotaSeleccionada ?? new Cuotas(); // Reutilizamos en edicion.
163
164 // Socio seleccionado: leemos desde SelectedItem para evitar problemas con ComboBox estilizado.
165 c.SocioId = (cmbSocio.SelectedItem as Socios)?.SocioId ?? 0;
166
167 // A�o (parseamos).
168 c.Anio = int.TryParse(txtAnio.Text, out int anio) ? anio : 0;
169
170 // Mes (parseamos).
171 c.Mes = int.TryParse(txtMes.Text, out int mes) ? mes : 0;
172
173 // Importe (aceptamos coma o punto decimal).
174 var importeTexto = (txtImporte.Text ?? string.Empty).Trim();
175 if (decimal.TryParse(importeTexto, NumberStyles.Any, CultureInfo.CurrentCulture, out var importe) ||
176 decimal.TryParse(importeTexto, NumberStyles.Any, CultureInfo.InvariantCulture, out importe))
177 {
178 c.Importe = importe; // Asignamos importe parseado.
179 }
180 else
181 {
182 c.Importe = -1; // Forzamos error de validacion en el controller.
183 }
184
185 // Fechas.
186 c.FechaVencimiento = dpVencimiento.SelectedDate ?? default(DateTime); // Vencimiento.
187 c.FechaPago = dpFechaPago.SelectedDate ?? default(DateTime); // Fecha pago (nullable).
188 c.Pagada = chkPagada.IsChecked == true; // Estado.
189
190 return c; // Devolvemos entidad.

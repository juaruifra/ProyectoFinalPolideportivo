161 private bool ValidarReserva(Reservas reserva, ref string tituloError, ref string mensajeError)
162 {

164

172
173 // El socio debe estar seleccionado.
174 if (ok && reserva.SocioId < 1)
175 {
176 tituloError = "Socio requerido"; // Titulo.
177 mensajeError = "Debe seleccionar un socio para la reserva."; // Mensaje.
178 ok = false; // Marcamos error.
179 }
180
181 // El socio debe estar activo en el club para poder reservar.
182 if (ok)
183 {

185 if (socio == null || !socio.Activo) // Si no existe o esta inactivo, rechazamos.
186 {

188 mensajeError = "El socio seleccionado no esta activo en el club y no puede realizar reservas."; // Mensaje.
189 ok = false; // Marcamos error.

193 // La instalacion debe estar seleccionada.
194 if (ok && reserva.InstalacionId < 1)

196 tituloError = "Instalacion requerida"; // Titulo.

198 ok = false; // Marcamos error.

200

202 if (ok)

205 if (instalacion == null || !instalacion.Disponible) // Si no existe o no esta disponible, rechazamos.
206 {
207 tituloError = "Instalacion no disponible"; // Titulo.
208 mensajeError = "La instalacion seleccionada no esta disponible para reservas."; // Mensaje.
209 ok = false; // Marcamos error.

211 }
212
